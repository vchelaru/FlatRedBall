{
    LayerProvidedByContainer = layerToAddTo;
    FlatRedBall.SpriteManager.AddPositionedObject(this);
    FlatRedBall.SpriteManager.AddLayer(BaseLayer);
    FlatRedBall.SpriteManager.MoveLayerAboveLayer(BaseLayer, LayerProvidedByContainer);
    FlatRedBall.SpriteManager.AddToLayer(BaseSprite, BaseLayer);
    AddToManagersBottomUp(layerToAddTo);
    CustomInitialize();
}
