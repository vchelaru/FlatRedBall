//==== Entities\Base ====
public virtual void AddToManagers (FlatRedBall.Graphics.Layer layerToAddTo) 
{
    LayerProvidedByContainer = layerToAddTo;
    FlatRedBall.SpriteManager.AddPositionedObject(this);
    FlatRedBall.SpriteManager.AddToLayer(BaseSprite, LayerProvidedByContainer);
    AddToManagersBottomUp(layerToAddTo);
    CustomInitialize();
}

// ---- AddToManagersBottomUp ----
public virtual void AddToManagersBottomUp (FlatRedBall.Graphics.Layer layerToAddTo) 
{
    AssignCustomVariables(false);
}

// ---- AssignCustomVariables ----
public virtual void AssignCustomVariables (bool callOnContainedElements) 
{
    if (callOnContainedElements)
    {
    }
    if (BaseSprite.Parent == null)
    {
        BaseSprite.Z = 1f;
    }
    else
    {
        BaseSprite.RelativeZ = 1f;
    }
}

//==== Entities\Derived ====
public override void AddToManagers (FlatRedBall.Graphics.Layer layerToAddTo) 
{
    LayerProvidedByContainer = layerToAddTo;
    FlatRedBall.SpriteManager.AddToLayer(BaseSprite, LayerProvidedByContainer);
    FlatRedBall.SpriteManager.AddToLayer(DerivedSprite, LayerProvidedByContainer);
    base.AddToManagers(layerToAddTo);
    CustomInitialize();
}

// ---- AddToManagersBottomUp ----
public override void AddToManagersBottomUp (FlatRedBall.Graphics.Layer layerToAddTo) 
{
    base.AddToManagersBottomUp(layerToAddTo);
}

// ---- AssignCustomVariables ----
public override void AssignCustomVariables (bool callOnContainedElements) 
{
    base.AssignCustomVariables(callOnContainedElements);
    if (callOnContainedElements)
    {
    }
    if (BaseSprite.Parent == null)
    {
        base.BaseSprite.Z = 9f;
    }
    else
    {
        base.BaseSprite.RelativeZ = 9f;
    }
    if (DerivedSprite.Parent == null)
    {
        DerivedSprite.Z = 1f;
    }
    else
    {
        DerivedSprite.RelativeZ = 1f;
    }
}
